// Interrupt callback function for BMP581/HDC302x and BNO086
void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin) {
    if(GPIO_Pin == BMP_INT_Pin) {
        BMP_Ready = 1;
    }
    if(GPIO_Pin == BNO_INT_Pin) {
        BNO_Ready = 1;
    }
    if(GPIO_Pin == HDC1_Ready) {
        HDC1_Ready = 1;
    }
    if(GPIO_Pin == HDC2_Ready) {
        HDC2_Ready = 1;
    }
    __HAL_GPIO_EXTI_CLEAR_IT(GPIO_Pin);
}
